# Lista de zonas sensibles SUSPENDIDA (decisión de producto, 2026-10-22).
# Mientras no haya datos reales de clientes, los PRs que tocan estas rutas no
# necesitan la etiqueta revisado-humano. Se conserva para reactivarla: antes de
# meter el primer dato real de un cliente (condición en docs/STATUS.md), copiar
# las líneas de las migraciones, RLS, auth, secretos y la propia guarda a
# sensitive-paths.txt. Los tests de la guarda la siguen usando para comprobar
# que, al reactivarla, funciona.
# Formato: <expresión regular extendida (grep -E) sobre la ruta><TAB><motivo>.
^packages/db/supabase/migrations/	migraciones
^packages/db/supabase/rollbacks/	rollbacks
\.sql$	SQL (políticas RLS, funciones, SECURITY DEFINER)
^packages/db/supabase/config\.toml$	config.toml de Supabase
^apps/web/src/proxy\.ts$	auth: proxy
^apps/web/src/lib/auth/	auth: DAL, sesión y redirecciones
^apps/web/src/lib/supabase/	auth: clientes de Supabase
^apps/web/src/app/login/	auth: login
^apps/web/src/app/app/actions\.ts$	auth: logout
^apps/web/src/app/app/page\.tsx$	auth: redirección tras el login (/app)
^apps/web/src/app/page\.tsx$	auth: redirección de la raíz
(^|/)\.env	secretos y variables de entorno
^apps/web/scripts/local-env\.ts$	secretos: local-env
^packages/db/src/seed/	secretos: seed (clave secreta local)
^packages/agent-core/.*(polic|autonom)	motor de políticas y techos de autonomía
^\.github/	CI y configuración de GitHub
^scripts/	guardas de mergeo y scripts del repo
(^|/)(vitest|playwright|eslint)\.config\.	configuración de tests y lint
(^|/)tsconfig[^/]*\.json$	configuración de TypeScript
^turbo\.json$	configuración de checks (turbo)
(^|/)postcss\.config\.	configuración de build
(^|/)next\.config\.	configuración de Next
(^|/)package\.json$	dependencias y scripts
^pnpm-lock\.yaml$	dependencias (lockfile)
^pnpm-workspace\.yaml$	dependencias (workspace)
^\.nvmrc$	versión de Node
(^|/)(AGENTS|CLAUDE)\.md$	reglas del repo (también en subcarpetas: las sesiones las cargan)
^docs/MANUAL\.md$	fuente de verdad (MANUAL)
(^|/)\.claude/	skills, agentes, permisos y hooks de las sesiones (también en subcarpetas)
(^|/)\.mcp\.json$	servidores MCP que cargan las sesiones
