## Anexo A: mergear un PR por REST con las comprobaciones de CLAUDE.md
```bash
#!/bin/bash
# merge-one.sh <pr> "<nota>": checks + sensitive-paths + check-sensitive de origin/main + comentario + squash con sha
set -u; R=dsvmotion/moodly; n=$1; note=$2; W=$(mktemp -d)
cd <ruta del clon> && git fetch -q origin main || exit 1
pr=$(gh api repos/$R/pulls/$n) || exit 1
sha=$(echo "$pr" | jq -r .head.sha); draft=$(echo "$pr" | jq -r .draft)
behind=$(gh api repos/$R/compare/main...$sha --jq .behind_by); [ "$behind" = 0 ] || { echo "behind main"; exit 1; }
ok=1; runs=$(gh api "repos/$R/commits/$sha/check-runs?per_page=100")
while IFS= read -r c; do [ -z "$c" ] && continue; case "$c" in \#*) continue;; esac
  r=$(echo "$runs" | jq -r --arg n "$c" '[.check_runs[]|select(.name==$n)]|sort_by(.started_at)|last|"\(.status) \(.conclusion)"')
  echo "  $c: $r"; [ "$r" = "completed success" ] || ok=0
done < <(git show origin/main:scripts/merge-guard/required-checks.txt)
st=$(gh api "repos/$R/commits/$sha/statuses?per_page=100" --jq '[.[]|select(.context=="sensitive-paths" and .creator.login=="github-actions[bot]")]|sort_by(.created_at)|last|.state')
[ "$st" = success ] || ok=0
gh api "repos/$R/pulls/$n/files?per_page=100" --paginate --jq '.[].filename' > $W/files
gh api -H "Accept: application/vnd.github.v3.diff" repos/$R/pulls/$n > $W/diff
git show origin/main:scripts/merge-guard/check-sensitive.sh > $W/cs.sh; git show origin/main:scripts/merge-guard/sensitive-paths.txt > $W/sp.txt
args=("$W/files" "$W/diff")
if grep -qx packages/evals/baseline.json $W/files; then git show origin/main:scripts/merge-guard/baseline-versions.sh > $W/bv.sh; bash $W/bv.sh $R main $sha $W || ok=0; args+=("$W/baseline-before.json" "$W/baseline-after.json"); fi
SENSITIVE_PATTERNS=$W/sp.txt bash $W/cs.sh "${args[@]}" || ok=0
[ $ok = 1 ] || { echo "#$n NOT OK"; exit 1; }
[ "$draft" = true ] && gh api -X POST repos/$R/pulls/$n/ccr/ready_for_review >/dev/null
gh api repos/$R/issues/$n/comments -f body="Mergeo. Head \`${sha:0:7}\`: checks obligatorios y \`sensitive-paths\` en success; check-sensitive.sh de origin/main limpio. $note

---
_Generated by [Claude Code](https://claude.ai/code)_" >/dev/null
gh api -X PUT repos/$R/pulls/$n/merge -f merge_method=squash -f sha=$sha
```
